CREATE ALGORITHM = UNDEFINED SQL SECURITY DEFINER VIEW v_order_detail AS  SELECT 
                of.order_detail_id,
                fv.img_food_url AS image,
                o.table_number,
                fl.food_name,
                v.variant_name,
                GROUP_CONCAT(fo.option_name SEPARATOR ',') AS option_name,
                of.quantity,
                of.base_price,
                ofs.serve_status_name,
                IFNULL(SUM(fo.option_price), 0) AS option_price,
    				((of.base_price + IFNULL(SUM(fo.option_price), 0)) * of.quantity) AS total_price
            FROM `order o`
            JOIN order_food of ON o.order_id = of.order_id
            JOIN food_variants fv ON of.food_variant_id = fv.food_variant_id
            JOIN food_list fl ON fv.food_id = fl.food_id
            JOIN food_type_ ft ON fl.food_type_id = ft.food_type_id
            LEFT JOIN variants v ON fv.variant_id = v.variant_id
            LEFT JOIN order_food_status ofs ON of.serve_status_id = ofs.serve_status_id
            LEFT JOIN order_food_options ofo ON of.order_detail_id = ofo.order_detail_id
            LEFT JOIN food_options fo ON ofo.options_id = fo.options_id
            WHERE DATE(of.date_time) = CURDATE() AND of.serve_status_id = 'N'
            GROUP BY of.order_detail_id
            ORDER BY of.date_time ASC ;

SELECT 
    o.order_id,
    o.date_time,
    t.table_number,
    o.total_price,
    os.order_status_name,
    CONCAT(e.emp_first_name, ' ', e.emp_last_name) AS emp_name,
    COUNT(ff.order_detail_id) AS total_items
FROM `order` o
JOIN table_no t ON t.table_number = o.table_number
JOIN order_status os ON os.order_status_id = o.order_status_id
JOIN employee e ON e.emp_id = o.emp_id
LEFT JOIN order_food ff ON ff.order_id = o.order_id
WHERE o.order_status_id = 'Y'
GROUP BY 
    o.order_id,
    o.date_time,
    t.table_number,
    o.total_price,
    os.order_status_name,
    e.emp_first_name,
    e.emp_last_name
ORDER BY o.date_time ASC ;

-- 1. เมื่อสั่งอาหารเพิ่ม (INSERT)
CREATE TRIGGER trg_after_order_food_insert
AFTER INSERT ON order_food
FOR EACH ROW
BEGIN
    UPDATE `order` o
    SET o.total_price = (
        SELECT IFNULL(SUM((of.base_price + IFNULL(opt.total_opt_price, 0)) * of.quantity), 0)
        FROM order_food of
        LEFT JOIN (
            SELECT ofo.order_detail_id, SUM(fo.option_price) AS total_opt_price
            FROM order_food_options ofo
            JOIN food_options fo ON ofo.options_id = fo.options_id
            GROUP BY ofo.order_detail_id
        ) opt ON of.order_detail_id = opt.order_detail_id
        WHERE of.order_id = NEW.order_id
    )
    WHERE o.order_id = NEW.order_id;
END //

END //
